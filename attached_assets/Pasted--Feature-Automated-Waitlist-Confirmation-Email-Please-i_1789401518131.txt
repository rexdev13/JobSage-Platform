3. Lead Submission Trigger (artifacts/api-server/src/routes/leads.ts)
In POST /api/leads/submit:
After saving the lead (where source === "form" and status === "new"):
Retrieve the saved lead ID.
Return HTTP 201 { success: true } without blocking on email delivery.
Asynchronously trigger the email dispatch in the background:
ts


void (async () => {
  try {
    const res = await sendWaitlistWelcomeEmail({
      to: d.email,
      firstName: d.firstName,
      industrySector: d.industrySector,
      desiredRole: d.desiredRole,
    });
    if (res.success) {
      await db
        .update(socialLeadsTable)
        .set({
          waitlistConfirmationSentAt: new Date(),
          waitlistConfirmationProviderId: res.messageId ?? null,
          waitlistConfirmationLastError: null,
        })
        .where(eq(socialLeadsTable.id, leadId));
    } else {
      await db
        .update(socialLeadsTable)
        .set({
          waitlistConfirmationLastError: res.error ?? "Failed to send",
        })
        .where(eq(socialLeadsTable.id, leadId));
    }
  } catch (err) {
    console.error("[leads] Background waitlist email dispatch failed:", err);
  }
})();
Ensure partial chat saves (POST /api/leads/chat) do not trigger this email.
